```typescript
// Core Template Interfaces
interface RecipeTemplate {
  id: string;
  name: string;
  description: string;
  createdAt: Date;
  updatedAt: Date;
  createdBy: string;
  customerId?: string;
  whiskeyTypeId: string;
  complexity: ComplexityLevel;
  tags: Tag[];
  mashBill: MashBillComponent[];
  isPublic: boolean;
  revisions: TemplateRevision[];
}

interface MashBillComponent {
  ingredientId: string;
  percentage: number;
  order: number;
}

interface Tag {
  id: string;
  name: string;
  category: TagCategory;
}

// Supporting Types
enum ComplexityLevel {
  BEGINNER = 'beginner',
  INTERMEDIATE = 'intermediate',
  ADVANCED = 'advanced'
}

enum TagCategory {
  FLAVOR_PROFILE = 'flavor_profile',
  PRODUCTION_METHOD = 'production_method',
  SPECIAL_TECHNIQUE = 'special_technique',
  CUSTOMER_SPECIFIC = 'customer_specific'
}

interface TemplateRevision {
  id: string;
  templateId: string;
  version: number;
  changes: string;
  createdAt: Date;
  createdBy: string;
  snapshot: RecipeTemplate;
}

// Search and Filter Types
interface TemplateSearchParams {
  query?: string;
  whiskeyTypes?: string[];
  tags?: string[];
  complexity?: ComplexityLevel[];
  customerId?: string;
  createdBy?: string;
  dateRange?: DateRange;
}

interface DateRange {
  startDate?: Date;
  endDate?: Date;
}

// Utility type for mash bill pattern recognition
interface MashBillPattern {
  pattern: string; // e.g. "75-21-00-04"
  ingredients: {
    corn?: number;
    rye?: number;
    wheat?: number;
    malted_barley?: number;
    other?: number;
  }
}
```
